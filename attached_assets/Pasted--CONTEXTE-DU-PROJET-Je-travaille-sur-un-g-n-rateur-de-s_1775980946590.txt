# Créer : src/GifRenderer.ts
# Voici EXACTEMENT ce que ce fichier doit faire :

ÉTAPE 1 — Initialisation du Canvas haute résolution
  - Créer un canvas NodeCanvas en 1200x400 (rendu 2x)
  - Appliquer ctx.scale(2, 2) pour résolution Retina
  - Créer un canvas de sortie 600x200 (taille finale signature)

ÉTAPE 2 — Instanciation de l'effet
  - Importer l'effet demandé via EffectFactory (voir ci-dessous)
  - Appeler effect.initialize(canvas, element)
  - L'élément (element) contient : x, y, width, height, content, opacity, rotation

ÉTAPE 3 — Boucle de rendu avec timing organique
  - Nombre de frames : 80 frames à 25 FPS
  - DeltaTime avec micro-variation naturelle :
      const jitter = 0.98 + Math.random() * 0.04;
      const deltaTime = (1000 / 25) * jitter;
  - À chaque frame : effect.update(deltaTime) → effect.render(ctx)
  - Downscale du canvas 2x vers 600x200 avant capture de frame

ÉTAPE 4 — Encodage GIF de qualité maximale
  - Utiliser gif-encoder-2 avec ces paramètres EXACTS :
      encoder.setQuality(1)      → qualité maximale
      encoder.setDither(true)    → Floyd-Steinberg dithering
      encoder.setRepeat(0)       → boucle infinie
      encoder.setDelay(40)       → 25 FPS (40ms par frame)
  - Seamless loop : démarrer le cycle de l'effet à son point naturel
    de début (t=0 du cycle physiologique) pour que frame 1 = frame 80

ÉTAPE 5 — Upload et cache
  - Calculer cacheKey = md5(userId + effectId + JSON.stringify(params))
  - Si cacheKey existe en DB → retourner l'URL directement (0ms)
  - Sinon → upload buffer GIF vers Cloudinary
  - Stocker en DB : { userId, cacheKey, gifUrl, params, createdAt }
  - Retourner l'URL Cloudinary

# Créer : src/EffectFactory.ts
# CE fichier est le SEUL endroit qui connaît les 55 effets
# Ne jamais mettre cette logique ailleurs

import { BreathingObjectEffect } from '../effects/breathing-object-effect';
// ... importer les 55 effets de la même façon

const EFFECTS_REGISTRY = {
  'organic-life-respiration-041': BreathingObjectEffect,
  // ... les 54 autres avec leur id exact comme clé
};

export function createEffect(effectId: string, config: any) {
  const EffectClass = EFFECTS_REGISTRY[effectId];
  if (!EffectClass) throw new Error(`Effet inconnu: ${effectId}`);
  return new EffectClass(config);
}

-- Créer : src/db/schema.sql

CREATE TABLE IF NOT EXISTS signatures_cache (
  id          SERIAL PRIMARY KEY,
  user_id     VARCHAR(255) NOT NULL,
  cache_key   VARCHAR(32) NOT NULL UNIQUE,  -- MD5 hash
  gif_url     TEXT NOT NULL,
  effect_id   VARCHAR(100),
  params      JSONB,
  created_at  TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_cache_key ON signatures_cache(cache_key);
CREATE INDEX idx_user_id   ON signatures_cache(user_id);
